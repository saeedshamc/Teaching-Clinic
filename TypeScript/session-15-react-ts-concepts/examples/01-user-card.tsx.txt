type UserCardProps = {
  name: string;
  age?: number;
  onSelect: (id: number) => void;
};

function UserCard({ name, age = 0, onSelect }: UserCardProps) {
  return (
    <button type="button" onClick={() => onSelect(1)}>
      {name} ({age})
    </button>
  );
}

export function Demo() {
  return <UserCard name="سارا" age={20} onSelect={(id) => console.log(id)} />;
}
